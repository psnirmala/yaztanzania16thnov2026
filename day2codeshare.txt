from pyspark.sql.types import *
from pyspark.sql import SparkSession
import time
spark=SparkSession.builder.appName("user-defined-schema-app").master("local[*]").getOrCreate()
fields=[StructField("id",IntegerType()),StructField("name",StringType()),
        StructField("designation",StringType())]
emp_schema=StructType(fields)
emp_df=spark.readStream.format("csv").schema(emp_schema).\
    load("/home/rps/training/structureddata/csvstreamingdata/")
dev_df=emp_df.where("designation='Developer'")
acct_df=emp_df.where("designation='Accountant'")
architect_df=emp_df.where("designation='Architect'")
dev_streaming_query=dev_df.writeStream.format("json").\
    option("checkpointLocation","/home/rps/training/checkpoint-dev"). \
    trigger(processingTime="20 seconds").start("/home/rps/streaming-out/dev")
acct_streaming_query=acct_df.writeStream.format("json").\
    option("checkpointLocation","/home/rps/training/checkpoint-acct"). \
    trigger(processingTime="20 seconds").start("/home/rps/streaming-out/acct")

architect_streaming_query=architect_df.writeStream.format("json").\
    option("checkpointLocation","/home/rps/training/checkpoint-architect"). \
    trigger(processingTime="20 seconds").start("/home/rps/streaming-out/architect")

print("streaming started")
time.sleep(5*60)
dev_streaming_query.stop()
acct_streaming_query.stop()
architect_streaming_query.stop()
print("streaming stopped")
