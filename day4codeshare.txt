from pyspark.sql import SparkSession

# 1. Create Spark Session
spark = (
    SparkSession.builder
    .appName("PySpark-Iceberg-Demo")
    .config(
        "spark.sql.extensions",
        "org.apache.iceberg.spark.extensions.IcebergSparkSessionExtensions"
    )
    .config(
        "spark.sql.catalog.local",
        "org.apache.iceberg.spark.SparkCatalog"
    )
    .config(
        "spark.sql.catalog.local.type",
        "hadoop"
    )
    .config(
        "spark.sql.catalog.local.warehouse",
        "file:///home/rps/iceberg_warehouse"
    )
    .config("spark.jars.packages",
            "org.apache.iceberg:iceberg-spark-runtime-4.0_2.13:1.10.0")
    .getOrCreate()
)

spark.sparkContext.setLogLevel("WARN")

# 2. Create sample student data
data = [
    (101, "Arun", "Python", 85),
    (102, "Bala", "Java", 90),
    (103, "Kumar", "PySpark", 95),
    (104, "Ravi", "Python", 80)
]

columns = ["student_id", "name", "course", "marks"]

df = spark.createDataFrame(data, columns)

print("Original Data:")
df.show()

# 3. Create Iceberg namespace
spark.sql("""
    CREATE NAMESPACE IF NOT EXISTS local.college
""")

# 4. Create Iceberg table
df.writeTo("local.college.students").create()

print("Iceberg table created successfully!")

# 5. Read Iceberg table
print("Reading Iceberg table:")
spark.read.table("local.college.students").show()

# 6. Insert additional records
new_data = [
    (105, "Suresh", "Spark", 88),
    (106, "Mani", "Python", 92)
]

new_df = spark.createDataFrame(new_data, columns)

new_df.writeTo("local.college.students").append()

print("After inserting new records:")
spark.read.table("local.college.students").show()

# 7. Execute SQL query
print("Students with marks above 90:")

spark.sql("""
    SELECT student_id, name, course, marks
    FROM local.college.students
    WHERE marks > 90
""").show()

# 8. Display Iceberg table history
print("Iceberg snapshots:")

spark.sql("""
    SELECT *
    FROM local.college.students.snapshots
""").show(truncate=False)

spark.stop()
